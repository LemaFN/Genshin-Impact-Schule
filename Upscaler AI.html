Supabase-Setup (einmalig)

Dieses SQL im Supabase-Dashboard unter „SQL Editor“ ausführen. Soll die Registrierung ohne Bestätigungs-Mail klappen: Authentication → Providers → Email → „Confirm email“ ausschalten.

create table if not exists public.hosts(
  user_id uuid not null references auth.users on delete cascade,
  device_id text not null, name text,
  last_seen timestamptz not null default now(),
  primary key(user_id, device_id));
create table if not exists public.jobs(
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users on delete cascade,
  host_device text not null, status text not null default 'queued',
  model text, width int, height int, progress int default 0,
  input_path text, output_path text, error text,
  created_at timestamptz not null default now());
alter table public.hosts enable row level security;
alter table public.jobs enable row level security;
create policy "hosts own" on public.hosts for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
create policy "jobs own" on public.jobs for all using (auth.uid()=user_id) with check (auth.uid()=user_id);
insert into storage.buckets(id,name,public) values('upscale','upscale',false) on conflict do nothing;
create policy "upscale own files" on storage.objects for all
  using (bucket_id='upscale' and (storage.foldername(name))[1]=auth.uid()::text)
  with check (bucket_id='upscale' and (storage.foldername(name))[1]=auth.uid()::text);
